        function analyzePassword(pwd) {
            const hasLower = /[a-z]/.test(pwd);
            const hasUpper = /[A-Z]/.test(pwd);
            const hasDigits = /\d/.test(pwd);
            const hasSymbols = /[^a-zA-Z0-9]/.test(pwd);

            let charsetSize = 0;
            if (hasLower) charsetSize += 26;
            if (hasUpper) charsetSize += 26;
            if (hasDigits) charsetSize += 10;
            if (hasSymbols) charsetSize += 32;

            charsetSize = Math.max(charsetSize, 1);
            const combinations = Math.pow(charsetSize, pwd.length);
            const speed = 1e9; // 1 bilhao de tentativas por segundo (GPU moderna)
            const seconds = combinations / speed;

            // Pontuacao de forca (0-100)
            let score = 0;
            score += pwd.length * 2;
            if (hasLower) score += 5;
            if (hasUpper) score += 10;
            if (hasDigits) score += 10;
            if (hasSymbols) score += 10;
            // Bonus por tamanho
            if (pwd.length >= 8) score += 10;
            if (pwd.length >= 12) score += 15;
            if (pwd.length >= 16) score += 10;
            // Bonus extra para senhas muito longas (20-40)
            if (pwd.length >= 25) score += 5;
            if (pwd.length >= 30) score += 5;
            
            score = Math.min(100, score);

            return {
                hasLower, hasUpper, hasDigits, hasSymbols,
                charsetSize, combinations, seconds, speed, score, length: pwd.length
            };
        }

        function resizeCanvas() {

        let canvasHistory = [];
        const MAX_HISTORY = 120;

        function drawCanvas(currentAttempt, maxAttempts) {
            const w = liveCanvas.width / window.devicePixelRatio;
            const h = liveCanvas.height / window.devicePixelRatio;

            canvasHistory.push(currentAttempt);
            if (canvasHistory.length > MAX_HISTORY) canvasHistory.shift();

            ctx.clearRect(0, 0, w, h);

            // Linhas de fundo
            ctx.strokeStyle = 'rgba(255,255,255,0.03)';
            ctx.lineWidth = 1;
            for (let y = 20; y < h; y += 20) {
                ctx.beginPath();
                ctx.moveTo(0, y);
                ctx.lineTo(w, y);
                ctx.stroke();
            }

            if (canvasHistory.length < 2) return;

            const maxVal = Math.max(...canvasHistory, 1);
            const step = w / (MAX_HISTORY - 1);

            // Preenchimento gradiente
            const gradient = ctx.createLinearGradient(0, 0, 0, h);
            gradient.addColorStop(0, 'rgba(255,107,43,0.15)');
            gradient.addColorStop(1, 'rgba(30,144,255,0.02)');

            ctx.beginPath();
            ctx.moveTo(0, h);
            for (let i = 0; i < canvasHistory.length; i++) {
                const x = i * step;
                const y = h - (canvasHistory[i] / maxVal) * (h - 20);
                if (i === 0) ctx.lineTo(x, y);
                else ctx.lineTo(x, y);
            }
            ctx.lineTo((canvasHistory.length - 1) * step, h);
            ctx.closePath();
            ctx.fillStyle = gradient;
            ctx.fill();

            // Linha
            ctx.beginPath();
            for (let i = 0; i < canvasHistory.length; i++) {
                const x = i * step;
                const y = h - (canvasHistory[i] / maxVal) * (h - 20);
                if (i === 0) ctx.moveTo(x, y);
                else ctx.lineTo(x, y);
            }
            ctx.strokeStyle = '#FF6B2B';
            ctx.lineWidth = 2;
            ctx.stroke();

            // Ponto final
            const lastX = (canvasHistory.length - 1) * step;
            const lastY = h - (canvasHistory[canvasHistory.length - 1] / maxVal) * (h - 20);
            ctx.beginPath();
            ctx.arc(lastX, lastY, 4, 0, Math.PI * 2);
            ctx.fillStyle = '#FF6B2B';
            ctx.fill();
            ctx.beginPath();
            ctx.arc(lastX, lastY, 8, 0, Math.PI * 2);
            ctx.fillStyle = 'rgba(255,107,43,0.2)';
            ctx.fill();
        }

                    <span class="b-label">Classificacao</span>
                    <span class="b-value" style="color: ${strengthLabel.color}">${strengthLabel.textContent}</span>
                </div>
            `;

            // Construir grade
            buildBruteGrid(10);
            const cells = bruteGrid.querySelectorAll('.brute-cell');

            // Charset para simulacao visual
            let charset = '';
            if (analysis.hasLower) charset += 'abcdefghijklmnopqrstuvwxyz';
            if (analysis.hasUpper) charset += 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
            if (analysis.hasDigits) charset += '0123456789';
            if (analysis.hasSymbols) charset += '!@#$%&*+-=?';
            if (!charset) charset = 'a';

            // Velocidade visual ajustada
            const visualSpeed = Math.min(50, Math.max(5, cells.length));
            let frame = 0;
            let simulatedAttempts = 0;
            // Posicao simulada na combinacao (0 a combinations-1)
            let currentPos = 0;

            function animate() {
                if (!isRunning) return;

                frame++;

                // Avancar posicao simulada
                const jumpPerFrame = Math.max(1, Math.floor(analysis.combinations / 500));
                currentPos = Math.min(currentPos + jumpPerFrame, analysis.combinations);

                // Atualizar celulas visuais
                for (let i = 0; i < cells.length; i++) {
                    if (frame % 3 === i % 3) {
                        const charIdx = Math.floor(Math.random() * charset.length);
                        cells[i].textContent = charset[charIdx];
                        cells[i].className = 'brute-cell scanning';
                    }
                }

                // Simular "match" parcial aleatorio
                if (frame % 8 === 0) {
                    const randomIdx = Math.floor(Math.random() * cells.length);
                    cells[randomIdx].className = 'brute-cell match';
                    cells[randomIdx].textContent = pwd[Math.floor(Math.random() * pwd.length)] || charset[0];
                }

                // Tentativa que "encontra" parcialmente
                if (frame % 15 === 0 && pwd.length > 0) {
                    const partialIdx = Math.floor(Math.random() * cells.length);
                    let partialText = '';
                    for (let c = 0; c < Math.min(pwd.length, 3); c++) {
                        partialText += Math.random() > 0.5 ? pwd[c] : charset[Math.floor(Math.random() * charset.length)];
                    }
                    cells[partialIdx].textContent = partialText;
                    cells[partialIdx].className = 'brute-cell match';
                }

                // Simular encontrou a senha ao final
                const progress = currentPos / analysis.combinations;
                if (progress >= 0.98 && frame > 60) {
                    // Encontrou!
                    const foundIdx = Math.floor(cells.length / 2);
                    cells[foundIdx].textContent = pwd.substring(0, 4);
                    cells[foundIdx].className = 'brute-cell found';

                    if (foundIdx + 1 < cells.length) {
                        cells[foundIdx + 1].textContent = pwd.substring(4, 8);
                        cells[foundIdx + 1].className = 'brute-cell found';
                        cells[foundIdx + 1].style.animationDelay = '0.1s';
                    }

                    progressFill.style.width = '100%';
                    progressAttempts.textContent = formatNumber(analysis.combinations) + ' tentativas';
                    progressPercent.textContent = '100%';

                    document.getElementById('statSpeed').textContent = formatNumber(analysis.speed);
                    drawCanvas(analysis.speed, analysis.speed);

                    showToast('Senha encontrada! (simulacao visual)', 'success');
                    isRunning = false;
                    attackBtn.classList.remove('running');
                    attackBtnText.textContent = 'Iniciar Simulacao';
                    attackBtn.querySelector('i').className = 'fas fa-crosshairs';
                    return;
                }

                // Atualizar progresso
                const displayProgress = Math.min(progress * 100, 99.9);
                progressFill.style.width = displayProgress + '%';
                simulatedAttempts = currentPos;
                progressAttempts.textContent = formatNumber(simulatedAttempts) + ' tentativas';
                progressPercent.textContent = displayProgress.toFixed(1) + '%';

                document.getElementById('statSpeed').textContent = formatNumber(analysis.speed);

                // Canvas
                const canvasVal = analysis.speed * (0.8 + Math.random() * 0.4);
                drawCanvas(canvasVal, analysis.speed * 1.5);

                animationId = requestAnimationFrame(animate);
            }

            animationId = requestAnimationFrame(animate);
            showToast('Simulacao iniciada - analisando combinacoes...', 'info');
        }
